cmake_minimum_required(VERSION 3.16)
project(FinalBuildFactions VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_factions src/factions.c)
add_library(fbs::factions ALIAS fbs_factions)
set_target_properties(fbs_factions PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME factions)
target_include_directories(fbs_factions PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_factions PUBLIC m)
  target_compile_options(fbs_factions PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_factions tests/test_factions.c)
  set_target_properties(fbs_test_factions PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_factions PRIVATE fbs::factions)
  if(NOT MSVC)
    target_compile_options(fbs_test_factions PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME factions COMMAND fbs_test_factions)
  set_tests_properties(factions PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_factions_example examples/basic.c)
  set_target_properties(fbs_factions_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_factions_example PRIVATE fbs::factions)
  if(BUILD_TESTING)
    add_test(NAME factions_example COMMAND fbs_factions_example)
  endif()
endif()
install(TARGETS fbs_factions EXPORT FinalBuildFactionsTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildFactionsTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildFactions)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-factions)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-factions/upstream OPTIONAL)
